const categoryLabel = { all: { en: "All", ar: "الكل" }, pain: { en: "Pain & fever", ar: "الألم والحمى" }, digestive: { en: "Digestive", ar: "الهضمي" }, antibiotic: { en: "Antibiotic", ar: "مضاد حيوي" }, allergy: { en: "Allergy", ar: "الحساسية" }, cardio: { en: "Cardio", ar: "القلب" }, metabolic: { en: "Metabolic", ar: "الأيض" }, respiratory: { en: "Respiratory", ar: "التنفس" } }; const state = { language: "en", category: "all", query: "", selectedId: (window.medicines && window.medicines[0]?.id) || null }; const refs = { medicineSearch: document.getElementById("medicineSearch"), searchResults: document.getElementById("searchResults"), medicineList: document.getElementById("medicineList"), medicineDetail: document.getElementById("medicineDetail"), entryCount: document.getElementById("entryCount"), languageToggle: document.getElementById("languageToggle"), filterButtons: [...document.querySelectorAll(".filter-btn")], translatedNodes: [...document.querySelectorAll("[data-en][data-ar]")] }; function getVisibleMedicines() { const query = state.query.trim().toLowerCase(); return medicines.filter((medicine) => { const matchesCategory = state.category === "all" || medicine.category === state.category; const searchableText = [ medicine.name, medicine.generic, ...medicine.brandNames, ...medicine.dosageForms, ...medicine.strengths, ...medicine.indications, ...medicine.warnings ].join(" ").toLowerCase(); const matchesQuery = !query || searchableText.includes(query); return matchesCategory && matchesQuery; }); } function renderFilters() { refs.filterButtons.forEach((button) => { button.classList.toggle("active", button.dataset.category === state.category); }); } function renderList() { if (!refs.medicineList || !refs.searchResults) return; const visibleMedicines = getVisibleMedicines(); const countText = visibleMedicines.length === 1 ? "1 result" : `${visibleMedicines.length} results`; refs.searchResults.textContent = state.language === "ar" ? `${visibleMedicines.length} ${visibleMedicines.length === 1 ? "نتيجة" : "نتائج"}` : countText; if (!visibleMedicines.length) { refs.medicineList.innerHTML = `
⌕

No medicines match

Try a different name or category.

`; refs.medicineDetail.innerHTML = `

Select a medicine to see details.

`; return; } const selectedExists = visibleMedicines.some((medicine) => medicine.id === state.selectedId); if (!selectedExists) { state.selectedId = visibleMedicines[0].id; } refs.medicineList.innerHTML = visibleMedicines .map((medicine) => { const isSelected = medicine.id === state.selectedId; const tag = categoryLabel[medicine.category][state.language]; return `
${tag} ${medicine.strengths[0]}

${medicine.name}

${medicine.generic}

${medicine.brandNames.join(", ")} `; }) .join(""); if (refs.medicineDetail) { renderDetail(); } } function renderDetail() { if (!refs.medicineDetail) return; const selected = medicines.find((medicine) => medicine.id === state.selectedId); if (!selected) { refs.medicineDetail.innerHTML = `

Select a medicine to see details.

`; return; } refs.medicineDetail.innerHTML = `

Selected medicine

${selected.name}

${categoryLabel[selected.category][state.language]}
Generic ${selected.generic}
Brand names ${selected.brandNames.join(", ")}
Forms ${selected.dosageForms.join(", ")}
Strength ${selected.strengths.join(", ")}

Indications

    ${selected.indications.map((item) => `
  • ${item}
    • `).join("")}

Warnings

    ${selected.warnings.map((item) => `
  • ${item}
    • `).join("")}

Common interactions

    ${selected.interactions.map((item) => `
  • ${item}
    • `).join("")}

Contraindications

    ${selected.contraindications.map((item) => `
  • ${item}
    • `).join("")}

Source note ${selected.source}

`; applyTranslations(); } function applyTranslations() { const translatedNodes = document.querySelectorAll("[data-en][data-ar]"); translatedNodes.forEach((node) => { const key = state.language === "ar" ? "ar" : "en"; node.textContent = node.dataset[key]; }); if (refs.medicineSearch) { refs.medicineSearch.placeholder = refs.medicineSearch.dataset[`placeholder-${state.language}`] || refs.medicineSearch.dataset.placeholderEn; } if (refs.languageToggle) { refs.languageToggle.textContent = state.language === "ar" ? "English" : "العربية"; } document.documentElement.lang = state.language; document.documentElement.dir = state.language === "ar" ? "rtl" : "ltr"; } function initHomePage() { if (!refs.medicineSearch || !refs.filterButtons.length) return; refs.medicineSearch.addEventListener("input", (event) => { state.query = event.target.value; renderList(); }); refs.filterButtons.forEach((button) => { button.addEventListener("click", () => { state.category = button.dataset.category; renderFilters(); renderList(); }); }); refs.languageToggle.addEventListener("click", () => { state.language = state.language === "en" ? "ar" : "en"; applyTranslations(); renderList(); }); refs.entryCount.textContent = medicines.length; applyTranslations(); renderFilters(); renderList(); } function initDetailPage() { const detailRoot = document.getElementById("detailRoot"); if (!detailRoot) return; const params = new URLSearchParams(window.location.search); const id = Number(params.get("id")); const selected = medicines.find((medicine) => medicine.id === id) || medicines[0]; if (!selected) { detailRoot.innerHTML = '

Medicine not found.

'; return; } const related = medicines.filter((medicine) => medicine.id !== selected.id).slice(0, 4); detailRoot.innerHTML = `

Selected medicine

${selected.name}

${selected.generic}

${categoryLabel[selected.category][state.language]}
Brand names${selected.brandNames.join(", ")}
Forms${selected.dosageForms.join(", ")}
Strength${selected.strengths.join(", ")}
Generic${selected.generic}

Indications

    ${selected.indications.map((i) => `
  • ${i}
    • `).join("")}

Warnings

    ${selected.warnings.map((i) => `
  • ${i}
    • `).join("")}

Common interactions

    ${selected.interactions.map((i) => `
  • ${i}
    • `).join("")}

Contraindications

    ${selected.contraindications.map((i) => `
  • ${i}
    • `).join("")}

Source note ${selected.source}

More medicines

${related.map((medicine) => ` ${medicine.name} ${medicine.generic} `).join("")}
`; applyTranslations(); } function init() { if (document.body.dataset.page === "detail") { initDetailPage(); } else { initHomePage(); } } document.addEventListener("DOMContentLoaded", init);
